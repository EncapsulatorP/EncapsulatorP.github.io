const microField = new Float32Array(1024);

function drawRG(t) {
  const c = byId('canvas-rg'), wrap = byId('wrap-rg');
  const w = wrap.clientWidth || 720, h = wrap.clientHeight || 390;
  if (c.width !== w || c.height !== h) { c.width = w; c.height = h; }
  const ctx = c.getContext('2d');
  ctx.fillStyle = '#0b0f14';
  ctx.fillRect(0, 0, w, h);

  const sc = Number(byId('rg-scale').value);
  const meta = RG_META[sc];
  const raw = byId('uncap').checked;
  const digs = [...visibleDigits].map((d) => towerHeight(d, raw) / 3);
  const dLen = digs.length || 1;
  const kR = Number(byId('gallery-knot-r').value);
  const kE = Number(byId('gallery-knot-e').value);
  const roots = knotAttractors(kR, kE);
  const knotAmp = roots.length === 2 ? roots[1] : 0.25;

  // Compute 32x32 Layer 0 discrete micro-field
  let s0 = 0, sq0 = 0;
  for (let y = 0; y < 32; y++) {
    for (let x = 0; x < 32; x++) {
      const hA = digs[(x + y * 3) % dLen];
      const hB = digs[(x * 5 + y * 7) % dLen];
      const wave = 0.5 + 0.5 * Math.sin(t * 1.15 + x * 0.42 - y * 0.37 + knotAmp * Math.cos(t * 0.8 + ((x * 7) ^ (y * 11)) * 0.22));
      const v = Math.max(0, Math.min(1, 0.42 * hA + 0.22 * hB + 0.36 * wave));
      microField[y * 32 + x] = v;
      s0 += v; sq0 += v * v;
    }
  }
  const var0 = Math.max(1e-9, sq0 / 1024 - (s0 / 1024) ** 2);

  // Render square instrument frame centered in canvas
  const blk = meta.block;
  const side = 32 / blk;
  const gridPx = Math.min(w - 40, h - 32);
  const ox = (w - gridPx) / 2;
  const oy = (h - gridPx) / 2;
  const cell = gridPx / side;

  ctx.strokeStyle = 'rgba(45,64,75,0.55)';
  ctx.lineWidth = 1;
  ctx.strokeRect(ox - 6, oy - 6, gridPx + 12, gridPx + 12);

  let sB = 0, sqB = 0;
  for (let by = 0; by < side; by++) {
    for (let bx = 0; bx < side; bx++) {
      let acc = 0;
      for (let dy = 0; dy < blk; dy++) {
        for (let dx = 0; dx < blk; dx++) acc += microField[(by * blk + dy) * 32 + bx * blk + dx];
      }
      const avg = acc / (blk * blk);
      sB += avg; sqB += avg * avg;
      const cx = ox + (bx + 0.5) * cell;
      const cy = oy + (by + 0.5) * cell;

      // Precision instrument palette: slate, sky-blue, mint, gold
      let color;
      if (avg < 0.34) color = '#4c6278';
      else if (avg < 0.54) color = '#79baff';
      else if (avg < 0.72) color = '#a0f8d9';
      else color = '#ffce88';

      if (blk === 1) {
        artDot(ctx, cx, cy, Math.max(1.8, cell * 0.36), color);
      } else if (blk === 8) {
        // Layer 3: macro-gradient coarse-graining
        const grad = ctx.createRadialGradient(cx, cy, cell * 0.08, cx, cy, cell * 0.78);
        grad.addColorStop(0, color);
        grad.addColorStop(1, 'rgba(11,15,20,0.45)');
        ctx.fillStyle = grad;
        ctx.fillRect(cx - cell * 0.48, cy - cell * 0.48, cell * 0.96, cell * 0.96);
      } else {
        const sz = cell * 0.86;
        ctx.fillStyle = color;
        ctx.fillRect(cx - sz / 2, cy - sz / 2, sz, sz);
      }
    }
  }

  const nB = side * side;
  const varB = Math.max(0, sqB / nB - (sB / nB) ** 2);
  const varRetained = sc === 0 ? 100 : Math.min(100, (varB / var0) * 100);
  // Information loss in bits: log2(1024 / nB) + variance attenuation term
  const bitsLost = sc === 0 ? 0 : Math.log2(1024 / nB) + Math.max(0, -Math.log2(Math.max(0.01, varRetained / 100)) * 0.5);

  byId('rg-top-scale').textContent = meta.scale;
  byId('rg-top-state').textContent = meta.state;
  byId('rg-subtitle').textContent = meta.desc;
  byId('rg-scale-out').textContent = `Layer ${sc} (${blk}×${blk})`;
  byId('rg-kpi-var').textContent = varRetained.toFixed(1) + '%';
  byId('rg-kpi-ent').textContent = bitsLost.toFixed(2) + ' bits';
  byId('rg-kpi-scale').textContent = `${meta.scale} · ${meta.state}`;
  byId('rg-kpi-seed').textContent = `${units.get(selected).label} · r=${kR.toFixed(2)}`;
}

let displayedDimension = null;
function dimensionState() {
  const unit = units.get(selected), digit = unit.whole.at(-1);
  const h = configuredHeight(digit, byId('uncap').checked, ruleConfig());
  const bias = Number(byId('dimension-bias').value), floor = Number(byId('dimension-floor').value);
  const driven = byId('dimension-drive').value === 'tower', detector = byId('dimension-view').value;
  const mu = bias + (driven ? 0.30 * (h - 2) : 0), radius = Math.sqrt(Math.max(0, mu));
  const active = radius > 0, resolved = active && radius > floor, visible = resolved && detector !== 'power';
  return { digit, h, bias, floor, driven, detector, mu, radius, active, resolved, visible, hole: visible && detector === 'iq' };
}
function renderDimension() {
  displayedDimension = dimensionState();
  const d = displayedDimension;
  byId('dimension-unit').value = selected;
  byId('dimension-bias-label').textContent = d.bias.toFixed(2);
  byId('dimension-floor-label').textContent = d.floor.toFixed(2);
  byId('dimension-kpis').replaceChildren(
    property('Attractor dimension', d.active ? 1 : 0),
    property('Visible dimension', d.visible ? 1 : 0),
    property('Visible holes', d.hole ? 1 : 0),
  );
  byId('dimension-caption').textContent = !d.active ? 'The loop has collapsed. The attracting state is a fixed point.'
    : !d.resolved ? 'The loop is below the detection floor. The oscillator still has a limit cycle.'
    : d.detector === 'sum' ? 'The mirror fields cancel in the imaginary quadrature. Their sum moves along a line.'
    : d.detector === 'power' ? 'The field keeps rotating. Its steady intensity is constant.'

  byId('dimension-command').textContent = 'Unit digit ' + d.digit + ' → height ' + d.h + ' · ' + (d.driven

  updateInvariants();
}
function paintDimension(timeMs) {
  const ctx = byId('dimension-canvas').getContext('2d');
  if (!ctx || !displayedDimension) return;
  const { radius, visible, detector, resolved } = displayedDimension;
  const x = 320, y = 122, angle = 0.0022 * timeMs + 0.66, px = radius * 88;
  ctx.clearRect(0, 0, 640, 244);
  ctx.strokeStyle = '#263c48'; ctx.lineWidth = 1;
  ctx.beginPath(); ctx.moveTo(55, y); ctx.lineTo(585, y); ctx.moveTo(x, 19); ctx.lineTo(x, 225); ctx.stroke();
  ctx.font = '14px system-ui'; ctx.fillStyle = '#abbcc4';
  ctx.fillText(detector === 'power' ? 'Power' : detector === 'sum' ? 'Real field sum' : 'In-phase field', 493, 145);
  if (detector === 'iq') ctx.fillText('Quadrature field', 330, 27);
  if (!visible) { artDot(ctx, detector === 'power' && resolved ? x + 106 * radius ** 2 : x, y, 5.5, '#ffce88'); return; }
  ctx.strokeStyle = '#a0f8d9'; ctx.lineWidth = 2.5; ctx.beginPath();
  if (detector === 'sum') { ctx.moveTo(x - 2 * px, y); ctx.lineTo(x + 2 * px, y); } else ctx.arc(x, y, px, 0, 2 * Math.PI);
  ctx.stroke();
  if (detector === 'sum') artDot(ctx, x + 2 * px * Math.cos(angle), y, 5.5, '#ffce88');
  else { artDot(ctx, x + px * Math.cos(angle), y - px * Math.sin(angle), 5.5, '#a0f8d9'); artDot(ctx, x + px * Math.cos(angle), y + px * Math.sin(angle), 5.5, '#d6c9ff'); }
